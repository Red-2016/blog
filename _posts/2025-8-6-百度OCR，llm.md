```go
package service

import (
	"encoding/base64"
	"encoding/json"
	"fin_statement_parser/app/dto"
	"fmt"
	"io"
	"log"
	"net/http"
	"net/url"
	"strings"
	"time"
)

const (
	OCR_URL   = "https://aip.baidubce.com/rest/2.0/ocr/v1/handwriting"
	TOKEN_URL = "https://aip.baidubce.com/oauth/2.0/token"
)

type OCRService struct {
	apiKey            string
	secretKey         string
	accessToken       string
	tokenExpiry       int64
	generateTokenTime int64
}

func NewOCRService(apiKey string, secretKey string) *OCRService {
	return &OCRService{apiKey: apiKey, secretKey: secretKey}
}

func (s *OCRService) getAccessToken() (string, error) {
	if s.accessToken != "" && s.tokenExpiry > 0 {
		// 检查token是否过期，提前5分钟更新token
		currentTime := time.Now().Unix()
		// 计算token剩余有效期（秒）
		remainTime := s.generateTokenTime + s.tokenExpiry - currentTime
		// 如果剩余时间大于5分钟（300秒），继续使用当前token
		if remainTime > 300 {
			return s.accessToken, nil
		}
		// 否则重新获取token
	}

	postData := fmt.Sprintf("grant_type=client_credentials&client_id=%s&client_secret=%s", s.apiKey, s.secretKey)
	resp, err := http.Post(TOKEN_URL, "application/x-www-form-urlencoded", strings.NewReader(postData))
	if err != nil {
		return "", fmt.Errorf("获取access_token失败: %v", err)
	}
	defer resp.Body.Close()

	body, err := io.ReadAll(resp.Body)
	if err != nil {
		return "", fmt.Errorf("读取access_token响应失败: %v", err)
	}

	var tokenResp map[string]interface{}
	if err := json.Unmarshal(body, &tokenResp); err != nil {
		return "", fmt.Errorf("解析access_token响应失败: %v", err)
	}

	token, ok := tokenResp["access_token"].(string)
	if !ok {
		return "", fmt.Errorf("access_token不存在或格式错误")
	}

	s.accessToken = token
	// 记录当前时间戳
	s.generateTokenTime = time.Now().Unix()
	// 保存过期时间
	if expiresIn, ok := tokenResp["expires_in"].(float64); ok {
		s.tokenExpiry = int64(expiresIn)
	} else {
		// 如果API没有返回过期时间，使用默认的30天(2592000秒)
		s.tokenExpiry = 2592000
	}
	return token, nil
}

func (s *OCRService) DecodeMultipleBase64ToBytes(base64List []string) ([][]byte, error) {
	var fileDatas [][]byte
	for _, base64Str := range base64List {
		fileData, err := base64.StdEncoding.DecodeString(base64Str)
		if err != nil {
			log.Println("base64解码失败:", err)
			return nil, fmt.Errorf("base64解码失败: %v", err)
		}
		fileDatas = append(fileDatas, fileData)
	}
	return fileDatas, nil
}

const (
	FILE_TYPE_IMAGE = 1
	FILE_TYPE_PDF   = 2
)

func (s *OCRService) RecognizeMultipleFilesByBytes(base64List []string, fileType int) (string, error) {
	fileDatas, err := s.DecodeMultipleBase64ToBytes(base64List)
	if err != nil {
		return "", err
	}
	var resultBuilder strings.Builder
	if fileType == FILE_TYPE_IMAGE {
		for i, data := range fileDatas {
			result, err := s.RecognizeImageByBytes(data)
			if err != nil {
				log.Printf("处理第 %d 张图片失败: %v", i+1, err)
				return "", fmt.Errorf("处理第 %d 张图片失败: %v", i+1, err)
			}
			resultBuilder.WriteString(result)
			if i < len(fileDatas)-1 {
				resultBuilder.WriteString("\n\n") // 在图片之间添加空行
			}
		}
		return resultBuilder.String(), nil
	} else if fileType == FILE_TYPE_PDF {
		for i, data := range fileDatas {
			result, err := s.RecognizePDFByBytes(data)
			if err != nil {
				log.Printf("处理第 %d 个PDF文件失败: %v", i+1, err)
				return "", fmt.Errorf("处理第 %d 个PDF文件失败: %v", i+1, err)
			}
			resultBuilder.WriteString(result)
			if i < len(fileDatas)-1 {
				resultBuilder.WriteString("\n\n")
			}
		}
		return resultBuilder.String(), nil
	} else {
		log.Println("不支持的文件类型")
		return "", fmt.Errorf("不支持的文件类型")
	}
}

// recognizeImage 使用byte识别单个图片中的文字
// 在调用函数前需要将base64 编码的字符串解码为原始字节数据
// imageData, err := base64.StdEncoding.DecodeString(imgStr)
func (s *OCRService) RecognizeImageByBytes(imageData []byte) (string, error) {
	token, err := s.getAccessToken()
	if err != nil {
		return "", err
	}

	requestURL := fmt.Sprintf("%s?access_token=%s", OCR_URL, token)

	// 对图片进行Base64编码
	imageBase64 := base64.StdEncoding.EncodeToString(imageData)

	// 构建请求参数
	params := url.Values{}
	params.Add("image", imageBase64)
	params.Add("detect_direction", "false")
	params.Add("probability", "false")
	params.Add("detect_alteration", "false")

	// 发送请求
	client := &http.Client{}
	req, err := http.NewRequest("POST", requestURL, strings.NewReader(params.Encode()))
	if err != nil {
		return "", fmt.Errorf("创建请求失败: %v", err)
	}

	req.Header.Add("Content-Type", "application/x-www-form-urlencoded")
	req.Header.Add("Accept", "application/json")

	// 发送请求并获取响应
	resp, err := client.Do(req)
	if err != nil {
		return "", fmt.Errorf("发送请求失败: %v", err)
	}
	defer resp.Body.Close()

	body, err := io.ReadAll(resp.Body)
	if err != nil {
		return "", fmt.Errorf("读取响应失败: %v", err)
	}

	return s.ParseOCRResponse(body)
}

// recognizePDF 识别PDF中的文字
func (s *OCRService) RecognizePDFByBytes(pdfData []byte) (string, error) {
	token, err := s.getAccessToken()
	if err != nil {
		return "", err
	}

	// 构建请求URL
	requestURL := fmt.Sprintf("%s?access_token=%s", OCR_URL, token)
	pdfBase64 := base64.StdEncoding.EncodeToString(pdfData)

	var resultBuilder strings.Builder
	 maxPages := 20

	// 循环处理每一页
	for pageNum := 1; pageNum <= maxPages; pageNum++ {
		params := url.Values{}
		params.Add("pdf_file", pdfBase64)
		params.Add("pdf_file_num", fmt.Sprintf("%d", pageNum)) // 指定页码
		params.Add("detect_direction", "false")
		params.Add("probability", "false")

		// 发送请求
		client := &http.Client{}
		req, err := http.NewRequest("POST", requestURL, strings.NewReader(params.Encode()))
		if err != nil {
			return "", fmt.Errorf("创建请求失败: %v", err)
		}

		req.Header.Add("Content-Type", "application/x-www-form-urlencoded")
		req.Header.Add("Accept", "application/json")

		resp, err := client.Do(req)
		if err != nil {
			return "", fmt.Errorf("发送请求失败: %v", err)
		}
		defer resp.Body.Close()

		body, err := io.ReadAll(resp.Body)
		if err != nil {
			return "", fmt.Errorf("读取响应失败: %v", err)
		}

		// 检查是否返回错误（可能是页码超出范围）
		var errResp map[string]interface{}
		if json.Unmarshal(body, &errResp) == nil {
			if errMsg, ok := errResp["error_msg"].(string); ok {
				// 如果错误信息包含页码相关错误，说明已经处理完所有页面
				if strings.Contains(errMsg, "页码") || strings.Contains(errMsg, "page") {
					break
				}
			}
		}
		pageResult, err := s.ParseOCRResponse(body)
		if err != nil {
			// 如果解析失败，可能是页码超出范围，停止处理
			break
		}
		resultBuilder.WriteString(pageResult)
		resultBuilder.WriteString("\n")
	}
	return resultBuilder.String(), nil
}

// parseOCRResponse 解析OCR API的响应
func (s *OCRService) ParseOCRResponse(responseData []byte) (string, error) {
	var ocrResp dto.BaiduOCRResponse
	if err := json.Unmarshal(responseData, &ocrResp); err != nil {
		var errResp map[string]interface{}
		if jsonErr := json.Unmarshal(responseData, &errResp); jsonErr == nil {
			if errMsg, ok := errResp["error_msg"].(string); ok {
				return "", fmt.Errorf("OCR API错误: %s", errMsg)
			}
		}
		return "", fmt.Errorf("解析OCR响应失败: %v, 响应内容: %s", err, string(responseData))
	}
	var result strings.Builder
	for _, word := range ocrResp.WordsResult {
		result.WriteString(word.Words)
	}
	return result.String(), nil
}
```

# LLM

```go
package service

import (
	"context"
	"encoding/json"
	"fmt"
	"log"

	"github.com/tmc/langchaingo/llms"
	"github.com/tmc/langchaingo/llms/openai"
	"github.com/tmc/langchaingo/outputparser"
)

type LLMService struct {
	llm *openai.LLM
}

func InitLLM(modelName string, apiUrl string, apiKey string) (*LLMService, error) {
	llm, err := openai.New(
		openai.WithModel(modelName),
		openai.WithBaseURL(apiUrl),
		openai.WithToken(apiKey),
	)
	if err != nil {
		log.Fatal(err)
		return nil, err
	}
	llmService := &LLMService{llm: llm}
	return llmService, nil
}

func (llmService *LLMService) GenerateAIResponse(
	ctx context.Context,
	input string,
	systemPrompt string,
	temperature float64,
	T any) (any, error) {

	parser, _ := outputparser.NewDefined(T)

	systemPromptWithParserStr := systemPrompt + parser.GetFormatInstructions()

	messages := []llms.MessageContent{
		llms.TextParts(llms.ChatMessageTypeSystem, systemPromptWithParserStr),
		llms.TextParts(llms.ChatMessageTypeHuman, input),
	}
	content, err := llmService.llm.GenerateContent(ctx, messages, llms.WithOptions(llms.CallOptions{Temperature: temperature}))
	if err != nil {
		log.Fatal(err)
	}
	log.Printf("AI响应内容: %s", content.Choices[0].Content)
	log.Printf("停止原因: %s", content.Choices[0].StopReason)

	result, err := parser.Parse(content.Choices[0].Content)
	if err != nil {
		log.Printf("parser解析JSON错误: %v", err)
		return "", fmt.Errorf("parser解析JSON错误: %v", err)
	}

	return result, nil
}

func AnyToStruct[T any](any interface{}) (T, error) {
	var t T
	jsonBytes, err := json.Marshal(any)
	if err != nil {
		return t, err
	}
	err = json.Unmarshal(jsonBytes, &t)
	return t, err
}
```
