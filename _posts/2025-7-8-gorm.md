# gorm

```go
package gorm

import (
	"fmt"
	"log"
	"os"
	"time"

	"gorm.io/driver/mysql"
	"gorm.io/gorm"
	"gorm.io/gorm/logger"
	"gorm.io/gorm/schema"
)

type Login struct {
	Username string `json:"username"`
	Password string `json:"password"`
}

func (Login) TableName() string {
	return "login"
}

func GormQuickStart() {
	host := "localhost"
	port := "3306"
	dbname := "test"
	user := "root"
	password := "Cptbtptp123456"

	// 构建连接字符串
	dsn := fmt.Sprintf("%s:%s@tcp(%s:%s)/%s?charset=utf8mb4&parseTime=True&loc=Local", user, password, host, port, dbname)

	// 连接数据库
	// gorm.Open(mysql.Open(dsn), nil) nil是默认配置，可以传入&gorm.Config{}设置配置项
	db, err := gorm.Open(mysql.Open(dsn), nil)
	if err != nil {
		panic(err)
	}

	// 写入数据
	instance1 := Login{Username: "test1", Password: "123456"}
	db.Create(&instance1)

	// 读取
	var instance2 Login
	db.Find(&instance2)
	fmt.Printf("%#v\n", instance2)
}

func CreateConnection(host, dbname, user, password string, port int) *gorm.DB {
	// 自定义日志输出，传入到下面的自定义配置的数据库连接
	newLogger := logger.New(
		log.New(os.Stdout, "\r\n", log.LstdFlags), // 输出到控制台
		logger.Config{
			SlowThreshold:        500 * time.Millisecond, // 超过500毫秒的SQL查询会被log记录
			LogLevel:             logger.Info,            // 输出SQL日志的级别，Slient为不输出
			ParameterizedQueries: true,                   // true代表SQL日志中不显示参数
			Colorful:             true,                   // 彩色打印
		},
	)

	dsn := fmt.Sprintf("%s:%s@tcp(%s:%s)/%s?charset=utf8mb4&parseTime=True&loc=Local", user, password, host, port, dbname)


	// 传入&gorm.Config{}设置配置项
	db, err := gorm.Open(mysql.Open(dsn), &gorm.Config{
		// 自定义命名
		NamingStrategy: schema.NamingStrategy{

			// 表名映射时不加复数s，例如User -> user不是users
			SingularTable: false,

			// 字段添加前缀，user -> t_user(s)
			TablePrefix: "t_",

			// 不需要驼峰转下划线
			NoLowerCase: false,
		},
		Logger: newLogger, // 使用上面自定义的日志输出
	})
	if err != nil {
		panic(err)
	}

	// 设置连接池控制参数
	sqlDB, _ := db.DB() // 获取sql.DB对象
	sqlDB.SetMaxIdleConns(10) // 设置连接池的最大空闲连接数，在空闲没有请求时，预先开10个连接，以备突然的请求
	sqlDB.SetMaxOpenConns(100) // 设置连接池的最大连接数
	sqlDB.SetConnMaxLifetime(time.Hour) // 设置连接的最大生命周期，超过这个时间没有被使用的连接会被关闭

	return db
}

type User struct {
	// gorm:"primaryKey" 表示字段是主键，创建时自增
	Id       int `gorm:"primaryKey"`
	Username string

	// column:xxx 规定列名
	UserId   string `gorm:"column:user_id"`
	Password string

	// 名称为CreatedAt时自动记录创建时间
	CreatedAt time.Time `gorm:"column:create_time"`

	// 名称为UpdatedAt时自动记录更新时间，type:date让他只精确到天
	UpdatedAt time.Time `gorm:"column:update_time;type:date"`

	// gorm:"-" 表示结构体需要，数据库中表不需要
	Province string   `gorm:"-"`
	Keywords []string `gorm:"json"` // gorm:"json"把切片存储为json格式，读出来时转换为切片
}
```
